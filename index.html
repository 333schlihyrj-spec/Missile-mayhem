-- ============================================ -- 1. CREATE PICTURES TABLE -- ============================================ CREATE TABLE IF NOT EXISTS public.pictures ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE, name TEXT NOT NULL, storage_path TEXT NOT NULL, is_secret BOOLEAN NOT NULL DEFAULT FALSE, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- ============================================ -- 2. ENABLE ROW LEVEL SECURITY -- ============================================ ALTER TABLE public.pictures ENABLE ROW LEVEL SECURITY; -- ============================================ -- 3. DATABASE SECURITY POLICIES -- ============================================ DROP POLICY IF EXISTS "Users can view own pictures" ON public.pictures; CREATE POLICY "Users can view own pictures" ON public.pictures FOR SELECT TO authenticated USING (auth.uid() = user_id); DROP POLICY IF EXISTS "Users can add own pictures" ON public.pictures; CREATE POLICY "Users can add own pictures" ON public.pictures FOR INSERT TO authenticated WITH CHECK (auth.uid() = user_id); DROP POLICY IF EXISTS "Users can update own pictures" ON public.pictures; CREATE POLICY "Users can update own pictures" ON public.pictures FOR UPDATE TO authenticated USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id); DROP POLICY IF EXISTS "Users can delete own pictures" ON public.pictures; CREATE POLICY "Users can delete own pictures" ON public.pictures FOR DELETE TO authenticated USING (auth.uid() = user_id); -- ============================================ -- 4. CREATE PRIVATE STORAGE BUCKET -- ============================================ INSERT INTO storage.buckets (id, name, public) VALUES ('pictures', 'pictures', FALSE) ON CONFLICT (id) DO NOTHING; -- ============================================ -- 5. STORAGE SECURITY POLICIES -- ============================================ DROP POLICY IF EXISTS "Users can view own picture files" ON storage.objects; CREATE POLICY "Users can view own picture files" ON storage.objects FOR SELECT TO authenticated USING ( bucket_id = 'pictures' AND (storage.foldername(name))[1] = auth.uid()::TEXT ); DROP POLICY IF EXISTS "Users can upload own picture files" ON storage.objects; CREATE POLICY "Users can upload own picture files" ON storage.objects FOR INSERT TO authenticated WITH CHECK ( bucket_id = 'pictures' AND (storage.foldername(name))[1] = auth.uid()::TEXT ); DROP POLICY IF EXISTS "Users can delete own picture files" ON storage.objects; CREATE POLICY "Users can delete own picture files" ON storage.objects FOR DELETE TO authenticated USING ( bucket_id = 'pictures' AND (storage.foldername(name))[1] = auth.uid()::TEXT );
